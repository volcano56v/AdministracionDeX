package org.example.logica;

import com.google.api.core.ApiFuture;
import com.google.cloud.firestore.*;
import org.example.modelo.Habitacion;

import java.util.List;
import java.util.concurrent.ExecutionException;

public class HabitacionCRUD {

    // ✅ Crear o agregar habitación
    public static void agregarHabitacion(Habitacion habitacion) throws ExecutionException, InterruptedException {
        Firestore db = ConexionFirebase.getDB();
        ApiFuture<WriteResult> future = db.collection("habitaciones")
                .document(String.valueOf(habitacion.getNumero()))
                .set(habitacion);
        future.get(); // Espera a que la operación termine
        System.out.println("✅ Habitación agregada correctamente: " + habitacion.getNumero());
    }

    // ✅ Listar todas las habitaciones
    public static void listarHabitaciones() throws ExecutionException, InterruptedException {
        Firestore db = ConexionFirebase.getDB();
        ApiFuture<QuerySnapshot> future = db.collection("habitaciones").get();
        List<QueryDocumentSnapshot> documentos = future.get().getDocuments();

        System.out.println("📋 Lista de habitaciones:");
        for (QueryDocumentSnapshot doc : documentos) {
            System.out.println(doc.getId() + " → " + doc.getData());
        }
    }

    // ✅ Leer habitación por número
    public static void obtenerHabitacion(int numero) throws ExecutionException, InterruptedException {
        Firestore db = ConexionFirebase.getDB();
        DocumentReference docRef = db.collection("habitaciones").document(String.valueOf(numero));
        DocumentSnapshot snapshot = docRef.get().get();

        if (snapshot.exists()) {
            Habitacion habitacion = snapshot.toObject(Habitacion.class);
            System.out.println("🏠 Habitación encontrada: " + habitacion);
        } else {
            System.out.println("⚠️ No existe la habitación con número " + numero);
        }
    }

    // ✅ Actualizar una habitación
    public static void actualizarHabitacion(Habitacion habitacion) throws ExecutionException, InterruptedException {
        Firestore db = ConexionFirebase.getDB();
        ApiFuture<WriteResult> future = db.collection("habitaciones")
                .document(String.valueOf(habitacion.getNumero()))
                .set(habitacion);
        future.get();
        System.out.println("🔄 Habitación actualizada: " + habitacion.getNumero());
    }

    // ✅ Eliminar habitación
    public static void eliminarHabitacion(int numero) throws ExecutionException, InterruptedException {
        Firestore db = ConexionFirebase.getDB();
        ApiFuture<WriteResult> future = db.collection("habitaciones")
                .document(String.valueOf(numero))
                .delete();
        future.get();
        System.out.println("🗑️ Habitación eliminada: " + numero);
    }
}
